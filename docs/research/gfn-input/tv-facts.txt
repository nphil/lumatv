== padblock
#!/bin/sh
# webosbrew boot hook: keep a USB-wired game controller invisible to apps.
#
# Netflix opens gamepad evdev nodes, and this pad (054c:09cc, a DualShock 4
# clone) never sends reports when it is wired only for charging, so its
# uninitialised axes read as a stick held full up-left. Unbinding it from its
# HID driver removes the input nodes; USB power is unaffected, so it still
# charges. Only HID bus 0003 (USB) is matched - the same pad over Bluetooth
# is bus 0005 and keeps working.
#
# usbhid's quirks parameter is read-only on this kernel, so a watcher covers
# re-plugs. Remove this file to uninstall.

IDS="054C:09CC"

detach() {
  for d in /sys/bus/hid/devices/0003:*; do
    [ -e "$d/driver" ] || continue
    id=${d##*/}; id=${id#0003:}; id=${id%.*}
    case " $IDS " in *" $id "*)
      echo "${d##*/}" > "$d/driver/unbind" 2>/dev/null ;;
    esac
  done
}

(
  while :; do
    detach
    sleep 3
  done
) &

exit 0
